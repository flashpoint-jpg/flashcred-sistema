document.getElementById('formProposta').addEventListener('submit', async (e) => { e.preventDefault(); const formData = new FormData(); formData.append('nome', document.getElementById('nome').value); formData.append('cpf', document.getElementById('cpf').value); // Altere de 'whatsapp' para 'telefone' para bater com o painel admin formData.append('telefone', document.getElementById('whatsapp').value); formData.append('nascimento', document.getElementById('nascimento').value); formData.append('endereco', document.getElementById('endereco').value); formData.append('numero', document.getElementById('numero').value); formData.append('cep', document.getElementById('cep').value); formData.append('valorSolicitado', document.getElementById('valorSolicitado').value); const fileInput = document.getElementById('comprovante'); if (fileInput.files[0]) { formData.append('comprovante', fileInput.files[0]); } try { const response = await fetch('/api/proposta/criar', { method: 'POST', body: formData }); const resultado = await response.json(); if (resultado.sucesso) { alert('Proposta enviada com sucesso!'); document.getElementById('formProposta').reset(); } else { alert('Erro: ' + resultado.mensagem); } } catch (err) { alert('Erro ao enviar proposta.'); } });
